% ===================== SETUP =====================
\begin{frame}{The question}
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      \textbf{Goal.} Given ModelSEED reactions, find the (reaction embedding,
      clustering algorithm) pair that groups reactions so each cluster is
      \emph{clean} with respect to its \textbf{EC number} --- same enzyme class
      together, different classes apart.

      \vspace{8pt}
      \textbf{Why it matters}
      \begin{itemize}\small
        \item EC \textbf{annotation transfer} to unlabelled reactions
        \item \textbf{Curation}: flag EC mis-assignments (outliers in a clean cluster)
        \item A reaction \textbf{similarity index} (FAISS) for metabolic modelling
      \end{itemize}
    \end{column}
    \begin{column}{0.40\textwidth}
      \begin{block}{Dataset}\small
        \begin{tabular}{@{}ll@{}}
          reactions & 13{,}721\\
          full EC (4-level) & 5{,}622\\
          EC3 subclasses & 261\\
          EC2 / EC1 & 68 / 7\\
        \end{tabular}\\[3pt]
        non-obsolete, full EC, all compounds have SMILES.
      \end{block}
      \footnotesize Primary metric: \textbf{EC3 AMI} (chance- \& count-corrected).
      Also EC1/EC2/full, homogeneity, purity, and kNN retrieval.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{What we surveyed --- 22 embeddings $\times$ 9 algorithms}
  \begin{columns}[T]
    \begin{column}{0.42\textwidth}
      \begin{block}{Embeddings (22), by representational family}\footnotesize
        \chip{cStruct}{\fStruct{Structural} (diff): ECFP2/4/6, FCFP4, AtomPair, RDKit, MACCS, MHFP, DRFP, Avalon, torsion}\\[3pt]
        \chip{cPhys}{\fPhys{PhysChem}: RDKit-217 + Mordred-1613 desc.}\\[3pt]
        \chip{cCGR}{\fCGR{Atom-mapped}: CGR (linear)}\\[3pt]
        \chip{cLearn}{\fLearn{Learned}: RXNFP, ChemBERTa, \textbf{REXzyme}}\\[3pt]
        \chip{cReducedFill}{\fReduced{Reduced-graph}: ErG (pharmacophore)}\quad
        \chip{cGNNFill}{\fGNN{Graph-GNN}: GIN}\\[3pt]
        \chip{c3D}{\fThreeD{3D geometry}: Uni-Mol}
      \end{block}
      {\scriptsize The last four (\fLearn{REXzyme}, \fReduced{ErG}, \fGNN{GIN},
      \fThreeD{Uni-Mol}) are genuinely \emph{new axes} --- Part III.}
    \end{column}
    \begin{column}{0.29\textwidth}
      \begin{block}{Clustering (9)}\footnotesize
        K-means, Agglomerative (avg / complete / Ward), Birch,
        \textbf{Leiden}, \textbf{Louvain}, HDBSCAN, Butina
      \end{block}
    \end{column}
    \begin{column}{0.29\textwidth}
      \begin{block}{Evaluation}\footnotesize
        Common 256-d SVD / cosine space.\\[3pt]
        \textbf{Clustering:} AMI, homogeneity, V-measure, purity at EC1/2/3/full.\\[3pt]
        \textbf{Retrieval:} leave-one-out kNN EC accuracy.
      \end{block}
    \end{column}
  \end{columns}
  \takeaway{One harness, every embedding $\times$ every algorithm, scored identically --- 800+ configurations.}
\end{frame}

% ===================== PART II =====================
\partdivider{Part II}{Bond-changes \& fusion}{the method that beats every single view}

\begin{frame}{Attacking full-EC: Condensed Graph of Reaction (CGR)}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}\small
      \textbf{Idea.} Merge reactant + product graphs into one via \textbf{atom
      mapping}; bonds carry a \emph{dynamic} label (formed / broken / order-changed).
      It encodes the \textbf{actual bond transformation} --- exactly what EC classifies.

      \vspace{4pt}
      \textbf{Pipeline.} RXNMapper (atom map) $\rightarrow$ chython (compose CGR)
      $\rightarrow$ fold to fingerprint bits.

      \vspace{4pt}
      Prior art (Varnek/Nugmanov; Heid \& Green 2022); the \emph{new} part is applying
      it to \textbf{EC clustering on ModelSEED} head-to-head.
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Fair subset (n=10{,}260)}\footnotesize
        \begin{tabular}{@{}lcc@{}}
          \toprule
          & EC3 AMI & \textbf{full-EC 1-NN}\\
          \midrule
          \fStruct{ECFP4-diff} & 0.628 & 0.490\\
          \fPhys{PhysChem} & \best{0.667} & 0.484\\
          \fCGR{\textbf{CGR}} & 0.613 & \best{0.543}\\
          \bottomrule
        \end{tabular}
      \end{block}
      \footnotesize \best{CGR wins full-EC retrieval} (its target!) but doesn't dominate
      EC3; capped at \bad{75\% coverage} (hard-to-map enzymatic reactions).
    \end{column}
  \end{columns}
  \takeaway{No single representation is best at everything --- each view has \textbf{complementary} strengths.}
\end{frame}

\begin{frame}{Our method --- multi-view fusion}
  \begin{columns}[T]
    \begin{column}{0.60\textwidth}
      \centering\includegraphics[width=\linewidth]{fig_triple_combos.png}
    \end{column}
    \begin{column}{0.40\textwidth}\small
      \textbf{Recipe} (unsupervised, no fitting): per-view normalize $\rightarrow$
      concat $\rightarrow$ SVD-256 $\rightarrow$ agglomerative-average.

      \vspace{4pt}
      All 35 triples from \{\fPhys{PhysChem}, \fStruct{ECFP4/DRFP/MHFP/AtomPair},
      \fCGR{CGR}, \fLearn{ChemBERTa}\} scored (top 12 shown; bars = EC3 AMI /
      retrieval accuracy).
      \begin{itemize}
        \item Best EC3 AMI \best{PhysChem+CGR+ChemBERTa = 0.71} --- beats every single view (0.667).
        \item \fCGR{CGR}-containing triples win \textbf{full-EC retrieval} (0.57--0.58).
        \item \best{PhysChem is the anchor} (in every top triple); even ``weak'' \fLearn{ChemBERTa} earns its place.
      \end{itemize}
      Tight band (0.68--0.71): the \emph{recipe}, not the exact triple, is what matters.
    \end{column}
  \end{columns}
  \takeaway{Fusion of complementary views is a genuine SOTA-beater --- \emph{before} any supervision.}
\end{frame}

% ===================== PART III =====================
\partdivider{Part III}{Do novel representations help?}{pharmacophore, learned-graph, 3D shape, biochemistry-native --- and why none moves the needle}

\begin{frame}{Four genuinely-orthogonal axes}
  \small We asked: does a representation that ``sees'' the reaction \emph{differently}
  carry EC signal the 2D fingerprints miss? Four new axes, each a different view:
  \vspace{4pt}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{\chip{cReducedFill}{\fReduced{ErG}} pharmacophore reduced-graph}\footnotesize
        RDKit ErG --- atoms abstracted to pharmacophore \emph{types} on a reduced
        topological graph. ``Same arrangement of features?''
      \end{block}
      \begin{block}{\chip{cGNNFill}{\fGNN{GIN}} pretrained 2D-graph GNN}\footnotesize
        snap-stanford context-prediction GIN --- learns on the molecular
        \emph{graph}, not a SMILES string.
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{\chip{c3D}{\fThreeD{Uni-Mol}} 3D molecular geometry}\footnotesize
        SE(3) transformer on generated \emph{conformers} --- the only view of true
        molecular \emph{shape} in the pool.
      \end{block}
      \begin{block}{\chip{cLearn}{\fLearn{REXzyme}} biochemistry-native reaction}\footnotesize
        335M T5 encoder trained on \textbf{32M RHEA reaction$\rightarrow$enzyme
        pairs} --- the exact biochemical domain, not USPTO synthesis.
      \end{block}
    \end{column}
  \end{columns}
  \takeaway{These are real, distinct \emph{representations} --- shape, pharmacophore, learned graph, enzyme-supervised language.}
\end{frame}

\begin{frame}{Result --- none beats the 2D plateau, and fusing them \emph{hurts}}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \centering\includegraphics[width=\linewidth]{fig_crossaxis_fusion.png}
    \end{column}
    \begin{column}{0.45\textwidth}\small
      Single-view EC3 AMI (fair subset), all \bad{under} the plateau:
      \fThreeD{Uni-Mol} 0.62 $>$ \fCGR{CGR} 0.61 $>$ \fReduced{ErG} 0.57 $>$
      \fGNN{GIN} 0.53.
      \begin{itemize}
        \item \fLearn{REXzyme} headline negative: biochem-native pretraining
          annotates \bad{worse than USPTO rxnfp} (0.78 vs 0.83).
        \item Fusing all four \bad{loses} to the Part II incumbent (0.698 vs
          \best{0.710}); adding the best one on top \bad{hurts} ($-0.004$).
        \item The two most label-aligned \emph{off-the-shelf} FPs also miss:
          \fStruct{FCFP4} 0.62 ties ECFP; \fPhys{Mordred} (1613 desc.) 0.59
          \bad{$<$} RDKit-217 --- more features hurt.
      \end{itemize}
      \textbf{Representational diversity $\neq$ task-relevant diversity.}
    \end{column}
  \end{columns}
  \takeaway{EC is a \textbf{bond-change} label; once descriptors+CGR+ChemBERTa capture that,
    shape / pharmacophore / graph / catalysis views are redundant. The plateau is the
    \emph{taxonomy}, not any encoder's ceiling.}
\end{frame}

\begin{frame}{Takeaways}
  \begin{enumerate}\small
    \item \fStruct{Structural difference fingerprints} + \textbf{agglomerative-average} are the strong, cheap baseline (EC3 AMI $\approx$0.63); \fLearn{learned reaction embeddings} \emph{underperform} them.
    \item \textbf{Retrieval beats clustering} for annotation: confidence-gated 15-NN gives EC3 at 0.90 accuracy over 76\% of reactions.
    \item \fCGR{CGR} (atom-mapped bond changes) wins the full-EC retrieval it was designed for, but is coverage-limited and not dominant alone.
    \item \textbf{Multi-view fusion} of complementary views (\fPhys{PhysChem}$\oplus$\fCGR{CGR}$\oplus$\fLearn{ChemBERTa}) \best{beats every individual method} --- EC3 AMI 0.71 --- unsupervised.
    \item \textbf{More exotic $\neq$ better}: 3D shape, pharmacophore, learned-graph, and a 335M biochemistry-native model add \bad{no EC signal} the 2D views lack (Part III).
    \item Next: \textbf{hierarchical-EC metric learning}, evaluated zero-shot on held-out EC subclasses.
  \end{enumerate}
\end{frame}
