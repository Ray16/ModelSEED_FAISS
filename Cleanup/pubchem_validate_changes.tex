\begin{frame}{Cross-Validation: xref and Name Lookups}

\textbf{xref cross-validation}: Compounds with both ChEBI and KEGG
identifiers are queried against both databases. Conflicting CIDs are flagged
as \texttt{XREF\_CONFLICT} and excluded from auto-correction.

\bigskip
\textbf{Multi-name cross-validation}: Up to 3 compound names are queried.
If different names return different PubChem CIDs, the compound is marked
\texttt{AMBIGUOUS} and excluded.

\bigskip
\begin{table}[h]
\centering
\begin{tabular}{lp{7cm}}
\toprule
\textbf{New Status} & \textbf{Meaning} \\
\midrule
XREF\_CONFLICT & ChEBI and KEGG xrefs point to different PubChem compounds \\
AMBIGUOUS      & Multiple compound names resolve to different PubChem CIDs \\
\bottomrule
\end{tabular}
\end{table}

\bigskip
\footnotesize
\textit{Both categories require manual review before any correction is applied.}

\end{frame}

\begin{frame}{Safety Features}

\begin{columns}[T]
\begin{column}{0.48\textwidth}
\textbf{Data Preservation}
\begin{itemize}
  \item Timestamped backup before every write
  \item Corrections log appended, not overwritten
  \item ISO~8601 timestamps on every entry
  \item Backup auto-removed if no changes
\end{itemize}

\bigskip
\textbf{Formula/Charge Consistency}
\begin{itemize}
  \item Recomputed from corrected InChI
  \item Uses BiochemPy InChI parser
  \item All rows for a compound updated
\end{itemize}
\end{column}

\begin{column}{0.48\textwidth}
\textbf{Cache Management}
\begin{itemize}
  \item SQLite cache for resumability
  \item \texttt{-{}-clear-cache} for fresh runs
  \item Rate-limited PubChem API access
\end{itemize}

\bigskip
\textbf{Output Reports}
\begin{itemize}\footnotesize
  \item \texttt{pubchem\_validation\_report.tsv}
  \item \texttt{pubchem\_mismatches.tsv}
  \item \texttt{pubchem\_protonation\_diffs.tsv}
  \item \texttt{pubchem\_corrections\_log.tsv}
\end{itemize}
\end{column}
\end{columns}

\end{frame}
